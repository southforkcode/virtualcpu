cmake_minimum_required(VERSION 3.20)
project(virtualcpu LANGUAGES CXX)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/decode_table.hpp
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/generate_decode.py ${CMAKE_CURRENT_BINARY_DIR}/decode_table.hpp
    DEPENDS doc/ISA_ENCODING.md tools/generate_decode.py tools/encoding.py VERBATIM)
add_library(virtualcpu src/cpu.cpp src/protected.cpp src/replay.cpp ${CMAKE_CURRENT_BINARY_DIR}/decode_table.hpp)
target_include_directories(virtualcpu PUBLIC src PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_compile_options(virtualcpu PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
add_executable(cpu_conformance tests/cpu_conformance.cpp)
target_link_libraries(cpu_conformance PRIVATE virtualcpu)
add_test(NAME cpu_conformance COMMAND cpu_conformance)
add_executable(b1_conformance tests/b1_conformance.cpp)
target_link_libraries(b1_conformance PRIVATE virtualcpu)
add_test(NAME b1_conformance COMMAND b1_conformance)
add_executable(branch_conformance tests/branch_conformance.cpp)
target_link_libraries(branch_conformance PRIVATE virtualcpu)
add_test(NAME branch_conformance COMMAND branch_conformance)
add_test(NAME branch_toolchain COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_branches.py)
add_executable(sp_conformance tests/sp_conformance.cpp)
target_link_libraries(sp_conformance PRIVATE virtualcpu)
add_test(NAME sp_conformance COMMAND sp_conformance)
add_test(NAME sp_toolchain COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_sp_offsets.py)
target_compile_definitions(cpu_conformance PRIVATE SOURCE_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
add_test(NAME example_listings COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/generate_example_listings.py --check)
add_test(NAME toolchain COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_toolchain.py)
file(GLOB KERNEL_IMAGES CONFIGURE_DEPENDS examples/kernel_*.asm)
add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/kernel_images.hpp
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/generate_kernel_fixtures.py ${CMAKE_CURRENT_BINARY_DIR}/kernel_images.hpp
    DEPENDS ${KERNEL_IMAGES} tools/generate_kernel_fixtures.py tools/isa.py tools/encoding.py doc/ISA_ENCODING.md VERBATIM)
add_custom_target(kernel_images DEPENDS ${CMAKE_CURRENT_BINARY_DIR}/kernel_images.hpp)
add_executable(protected_conformance tests/protected_conformance.cpp)
add_dependencies(protected_conformance kernel_images)
target_include_directories(protected_conformance PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(protected_conformance PRIVATE virtualcpu)
add_test(NAME protected_conformance COMMAND protected_conformance)
add_executable(kernel_demo examples/kernel.cpp)
add_dependencies(kernel_demo kernel_images)
target_include_directories(kernel_demo PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(kernel_demo PRIVATE virtualcpu)
add_executable(console_demo examples/console.cpp)
add_dependencies(console_demo kernel_images)
target_include_directories(console_demo PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(console_demo PRIVATE virtualcpu)
add_test(NAME console_demo COMMAND console_demo --record ${CMAKE_CURRENT_BINARY_DIR}/console-replay.jsonl)
set_tests_properties(console_demo PROPERTIES FIXTURES_SETUP console_recording)
add_test(NAME console_replay COMMAND console_demo --replay ${CMAKE_CURRENT_BINARY_DIR}/console-replay.jsonl)
set_tests_properties(console_replay PROPERTIES FIXTURES_REQUIRED console_recording)
add_executable(console_conformance tests/console_conformance.cpp)
add_dependencies(console_conformance kernel_images)
target_include_directories(console_conformance PRIVATE ${CMAKE_CURRENT_BINARY_DIR} examples)
target_link_libraries(console_conformance PRIVATE virtualcpu)
add_test(NAME console_conformance COMMAND console_conformance)
file(GLOB COMPILER_CORPUS CONFIGURE_DEPENDS examples/workloads/*.c examples/workloads/*.asm)
add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/compiler/compiler_images.hpp ${CMAKE_CURRENT_BINARY_DIR}/compiler/compiler_images.json
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/generate_compiler_fixtures.py ${CMAKE_CURRENT_BINARY_DIR}/compiler/compiler_images.hpp
    DEPENDS ${COMPILER_CORPUS} ${KERNEL_IMAGES} examples/workloads/corpus.json tools/smallc.py tools/isa.py tools/encoding.py
            tools/generate_compiler_fixtures.py tools/generate_kernel_fixtures.py tools/measure_workloads.py
            tests/compiler_conformance.cpp tests/test_compiler.py tools/evaluate_pointer_arithmetic.py tests/test_pointer_evaluation.py
            examples/console_guest.hpp CMakeLists.txt src/cpu.cpp src/cpu.hpp src/protected.cpp src/replay.cpp doc/ISA_ENCODING.md VERBATIM)
add_custom_target(compiler_images DEPENDS ${CMAKE_CURRENT_BINARY_DIR}/compiler/compiler_images.hpp ${CMAKE_CURRENT_BINARY_DIR}/compiler/compiler_images.json)
add_executable(compiler_conformance tests/compiler_conformance.cpp)
add_dependencies(compiler_conformance kernel_images compiler_images)
target_include_directories(compiler_conformance PRIVATE ${CMAKE_CURRENT_BINARY_DIR} ${CMAKE_CURRENT_BINARY_DIR}/compiler examples)
target_link_libraries(compiler_conformance PRIVATE virtualcpu)
add_test(NAME compiler_conformance COMMAND compiler_conformance ${CMAKE_CURRENT_BINARY_DIR}/compiler/execution.json)
add_test(NAME compiler_frontend COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_compiler.py)
set_tests_properties(compiler_conformance PROPERTIES FIXTURES_SETUP compiler_execution)
add_test(NAME compiler_measurements COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/measure_workloads.py
    --build ${CMAKE_CURRENT_BINARY_DIR} --execution ${CMAKE_CURRENT_BINARY_DIR}/compiler/execution.json
    --output ${CMAKE_CURRENT_BINARY_DIR}/compiler/measurements.json)
set_tests_properties(compiler_measurements PROPERTIES FIXTURES_REQUIRED compiler_execution)
add_test(NAME pointer_model COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_pointer_evaluation.py)
add_test(NAME pointer_evaluation COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/evaluate_pointer_arithmetic.py
    --build ${CMAKE_CURRENT_BINARY_DIR}
    --output ${CMAKE_CURRENT_BINARY_DIR}/compiler/pointer-evaluation.json)
set_tests_properties(pointer_evaluation PROPERTIES FIXTURES_REQUIRED compiler_execution)
